type dNode struct {
        data string
        prev *dNode
        next *dNode
}

type DoublyList struct {
        head *dNode
        tail *dNode
        size int
}

func NewDoublyList() *DoublyList { return &DoublyList{} }

// вспомогательный поиск узла 
func (l *DoublyList) findNode(val string) *dNode {
        for curr := l.head; curr != nil; curr = curr.next {
                if curr.data == val {
                        return curr
                }
        }
        return nil
}

func (l *DoublyList) PushFront(val string) {
        node := &dNode{data: val, next: l.head}
        if l.head != nil {
                l.head.prev = node
        } else {
                l.tail = node
        }
        l.head = node
        l.size++
}

func (l *DoublyList) PushBack(val string) {
        node := &dNode{data: val, prev: l.tail}
        if l.tail != nil {
                l.tail.next = node
        } else {
                l.head = node
        }
        l.tail = node
        l.size++
}

// вставка ДО целевого элемента 
func (l *DoublyList) InsertBefore(target, val string) bool {
        curr := l.findNode(target)
        if curr == nil {
                return false
        }
        if curr == l.head {
                l.PushFront(val)
                return true
        }
        node := &dNode{data: val, prev: curr.prev, next: curr}
        curr.prev.next = node
        curr.prev = node
        l.size++
        return true
}

// вставка ПОСЛЕ целевого элемента 
func (l *DoublyList) InsertAfter(target, val string) bool {
        curr := l.findNode(target)
        if curr == nil {
                return false
        }
        if curr == l.tail {
                l.PushBack(val)
                return true
        }
        node := &dNode{data: val, prev: curr, next: curr.next}
        curr.next.prev = node
        curr.next = node
        l.size++
        return true
}

func (l *DoublyList) PopFront() error {
        if l.head == nil {
                return errors.New("list is empty")
        }
        l.head = l.head.next
        if l.head != nil {
                l.head.prev = nil
        } else {
                l.tail = nil
        }
        l.size--
        return nil
}

func (l *DoublyList) PopBack() error {
        if l.tail == nil {
                return errors.New("list is empty")
        }
        l.tail = l.tail.prev
        if l.tail != nil {
                l.tail.next = nil
        } else {
                l.head = nil
        }
        l.size--
        return nil
}

// удаление ДО целевого элемента 
func (l *DoublyList) RemoveBefore(target string) bool {
        curr := l.findNode(target)
        if curr == nil || curr.prev == nil {
                return false
        }
        toDel := curr.prev
        if toDel == l.head {
                _ = l.PopFront()
                return true
        }
        toDel.prev.next = curr
        curr.prev = toDel.prev
        l.size--
        return true
}

// удаление ПОСЛЕ целевого элемента 
func (l *DoublyList) RemoveAfter(target string) bool {
        curr := l.findNode(target)
        if curr == nil || curr.next == nil {
                return false
        }
        toDel := curr.next
        if toDel == l.tail {
                _ = l.PopBack()
                return true
        }
        curr.next = toDel.next
        toDel.next.prev = curr
        l.size--
        return true
}

func (l *DoublyList) RemoveByValue(val string) bool {
        curr := l.findNode(val)
        if curr == nil {
                return false
        }
        if curr == l.head {
                _ = l.PopFront()
                return true
        }
        if curr == l.tail {
                _ = l.PopBack()
                return true
        }
        curr.prev.next = curr.next
        curr.next.prev = curr.prev
        l.size--
        return true
}

        return l.findNode(val) != nil

type hashNode struct {
        key   string
        value string
        next  *hashNode
}

type HashTable struct {
        buckets  []*hashNode
        capacity int
        size     int
}

func NewHashTable() *HashTable {
        cap := 128
        return &HashTable{
                buckets:  make([]*hashNode, cap),
                capacity: cap,
        }
}

func (h *HashTable) hashFunction(key string) int {
        var hash uint64 = 0
        for i := 0; i < len(key); i++ {
                hash = hash*31 + uint64(key[i])
        }
        return int(hash % uint64(h.capacity))
}

func (h *HashTable) Insert(key, value string) {
        idx := h.hashFunction(key)
        curr := h.buckets[idx]
        for curr != nil {
                if curr.key == key {
                        curr.value = value
                        return
                }
                curr = curr.next
        }
        h.buckets[idx] = &hashNode{key: key, value: value, next: h.buckets[idx]}
        h.size++
}

func (h *HashTable) Get(key string) (string, error) {
        idx := h.hashFunction(key)
        curr := h.buckets[idx]
        for curr != nil {
                if curr.key == key {
                        return curr.value, nil
                }
                curr = curr.next
        }
        return "", errors.New("key not found")
}

func (h *HashTable) Contains(key string) bool {
        _, err := h.Get(key)
        return err == nil
}

func (h *HashTable) Remove(key string) bool {
        idx := h.hashFunction(key)
        curr := h.buckets[idx]
        var prev *hashNode
        for curr != nil {
                if curr.key == key {
                        if prev == nil {
                                h.buckets[idx] = curr.next
                        } else {
                                prev.next = curr.next
                        }
                        h.size--
                        return true
                }
                prev = curr
                curr = curr.next
        }
        return false
}

func (h *HashTable) Size() int { return h.size }

func (h *HashTable) toMap() map[string]string {
        m := make(map[string]string, h.size)
        for i := 0; i < h.capacity; i++ {
                for curr := h.buckets[i]; curr != nil; curr = curr.next {
                        m[curr.key] = curr.value
                }
        }
        return m
}

func (h *HashTable) SerializeText(filename string) error {
        bytes, err := json.MarshalIndent(h.toMap(), "", "  ")
        if err != nil {
                return err
        }
        return os.WriteFile(filename, bytes, 0644)
}

func (h *HashTable) DeserializeText(filename string) error {
        bytes, err := os.ReadFile(filename)
        if err != nil {
                return err
        }
        var m map[string]string
        if err := json.Unmarshal(bytes, &m); err != nil {
                return err
        }
        h.buckets = make([]*hashNode, h.capacity)
        h.size = 0
        for k, v := range m {
                h.Insert(k, v)
        }
        return nil
}

func (h *HashTable) SerializeBinary(filename string) error {
        f, err := os.Create(filename)
        if err != nil {
                return err
        }
        defer f.Close()
        return gob.NewEncoder(f).Encode(h.toMap())
}

func (h *HashTable) DeserializeBinary(filename string) error {
        f, err := os.Open(filename)
        if err != nil {
                return err
        }
        defer f.Close()
        var m map[string]string
        if err := gob.NewDecoder(f).Decode(&m); err != nil {
                return err
        }
        h.buckets = make([]*hashNode, h.capacity)
        h.size = 0
        for k, v := range m {
                h.Insert(k, v)
        }
        return nil
}

        head *sNode
        tail *sNode
        size int

func NewSinglyList() *SinglyList { return &SinglyList{} }

func (l *SinglyList) PushFront(val string) {
        node := &sNode{data: val, next: l.head}
        l.head = node
        if l.tail == nil {
                l.tail = node
        }
        l.size++
}

func (l *SinglyList) PushBack(val string) {
        node := &sNode{data: val}
        if l.tail == nil {
                l.head = node
                l.tail = node
        } else {
                l.tail.next = node
                l.tail = node
        }
        l.size++
}

// вставка ДО целевого элемента 
func (l *SinglyList) InsertBefore(target, val string) bool {
        if l.head == nil {
                return false
        }
        if l.head.data == target {
                l.PushFront(val)
                return true
        }
        curr := l.head
        for curr.next != nil && curr.next.data != target {
                curr = curr.next
        }
        if curr.next == nil {
                return false
        }
        curr.next = &sNode{data: val, next: curr.next}
        l.size++
        return true
}

// вставка ПОСЛЕ целевого элемента 
func (l *SinglyList) InsertAfter(target, val string) bool {
        curr := l.head
        for curr != nil && curr.data != target {
                curr = curr.next
        }
        if curr == nil {
                return false
        }
        node := &sNode{data: val, next: curr.next}
        curr.next = node
        if curr == l.tail {
                l.tail = node
        }
        l.size++
        return true
}

        if l.head == nil {
                return errors.New("list is empty")

        if l.head == nil {
                l.tail = nil
        }
        l.size--
        return nil
}

        if l.head == nil {
                return errors.New("list is empty")
        }
        if l.head == l.tail {
                return l.PopFront()
        }
        curr := l.head
        for curr.next != l.tail {
                curr = curr.next
        }
        l.tail = curr
        l.tail.next = nil
        l.size--
        return nil
}
